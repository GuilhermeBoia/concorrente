## BARREIRA REUTILIZÁVEL

reusable_barrier {
    main() {
        count = 0
        Semaphore catraca = new Semaphore(0)
        Semaphore catraca2 = new Semaphore(1)
        Semaphore mutex = new Semaphore(1)
    }

    await() {
        
        mutex.wait()
        count ++
        if count == n:
            catraca2.wait()
            catraca.signal()
        mutex.signal()

        catraca.wait()
        catraca.signal()

        mutex.wait()
        count --
        if count == 0:
            catraca.wait()
            catraca2.signal()
        mutex.signal()

        catraca2.wait()
        catraca2.signal()

    }
}

## PRODUTOR CONSUMIDOR

main {
    SC = new Semaphore(0)
    SP = new Semaphore(k) //buffer size
    mutex = new Semaphore(1)
}

produtor {
   
    SP.wait()

    r = random_produce
    mutex.wait()
    produce(r)
    mutex.signal()

    SC.signal()
}

consumidor {
    SC.wait()
    
    mutex.wait()
    consume()
    mutex.signal()

    SP.signal()
}
